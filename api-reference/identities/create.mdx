---
title: "Create Agent Identity"
api: "POST https://api.wirebox.sh/api/v1/identities"
description: "Register a new autonomous agent identity and automatically provision a dedicated mailbox."
---

### Body Parameters

<ParamField body="agent_handle" type="string" required>
  Unique handle for this agent: 3–63 characters, lowercase letters, digits, and hyphens only; must start and end with a letter or digit; consecutive hyphens are not allowed. Example: `eva`.
</ParamField>

<ParamField body="display_name" type="string">
  Human-readable name for display in email headers and the Console. Example: `Eva Support`.
</ParamField>

<ParamField body="description" type="string">
  Optional context describing this agent's purpose.
</ParamField>

<ParamField body="imessage_enabled" type="boolean" default="false">
  Enable the Apple iMessage router channel for this identity.
</ParamField>

<ParamField body="custom_domain_id" type="string">
  Optional ID of a verified custom domain to use for this identity's mailbox address. Takes precedence over `domain`.
</ParamField>

<ParamField body="domain" type="string">
  Optional custom domain hostname to use for this identity's mailbox address. If neither `custom_domain_id` nor `domain` is provided, the organization's verified default domain is used when available.
</ParamField>

### Response

<ResponseField name="id" type="string">
  Unique identity identifier (e.g. `ident_01J8ABC123XYZ`).
</ResponseField>

<ResponseField name="agent_handle" type="string">
  The claimed handle.
</ResponseField>

<ResponseField name="display_name" type="string">
  Display name for the agent.
</ResponseField>

<ResponseField name="description" type="string">
  Description of the agent.
</ResponseField>

<ResponseField name="status" type="string">
  Identity status (`active`).
</ResponseField>

<ResponseField name="imessage_enabled" type="boolean">
  Whether the iMessage router channel is enabled.
</ResponseField>

<ResponseField name="mail_filter_mode" type="string">
  Legacy combined mail filter posture (`whitelist` or `blacklist`).
</ResponseField>

<ResponseField name="mail_inbound_filter_mode" type="string">
  Inbound guardrail posture: `whitelist` (only approved senders) or `blacklist` (open to anyone).
</ResponseField>

<ResponseField name="mail_outbound_filter_mode" type="string">
  Outbound guardrail posture: `whitelist` (only approved recipients) or `blacklist` (open to anyone).
</ResponseField>

<ResponseField name="email_address" type="string">
  The provisioned mailbox address (e.g. `eva@wireboxmail.com`).
</ResponseField>

<ResponseField name="mailbox" type="object">
  Provisioned mailbox summary: `id`, `email_address`, `created_at`.
</ResponseField>

<ResponseField name="tunnel" type="object">
  Provisioned network tunnel: `id`, `public_url`, `status`, `is_connected`.
</ResponseField>

<ResponseField name="created_at" type="string">
  ISO 8601 creation timestamp.
</ResponseField>

<ResponseField name="updated_at" type="string">
  ISO 8601 update timestamp.
</ResponseField>

<RequestExample>
```bash cURL
curl -X POST https://api.wirebox.sh/api/v1/identities \
  -H "Authorization: Bearer $WIREBOX_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "agent_handle": "eva",
    "display_name": "Eva Customer Support",
    "description": "Customer triage agent"
  }'
```

```python Python
from wirebox import Wirebox

with Wirebox() as wirebox:
    identity = wirebox.create_identity(
        "eva",
        display_name="Eva Customer Support",
        description="Customer triage agent"
    )
```

```ts TypeScript
import { Wirebox } from "@wirebox-sh/sdk";

const wirebox = new Wirebox();
const identity = await wirebox.createIdentity({
  agent_handle: "eva",
  display_name: "Eva Customer Support",
  description: "Customer triage agent",
});
```

```bash CLI
wirebox identity create \
  --handle eva \
  --display-name "Eva Customer Support" \
  --description "Customer triage agent"
```
